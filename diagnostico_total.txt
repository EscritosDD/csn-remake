--- parseOBJ3 ---
function parseOBJ3(txt,skin){
  const P=[],UV=[],idxMap={};
  const pos=[],uv=[],si=[],sw=[],outIdx=[];
  txt.split('\n').forEach(line=>{
    const p=line.trim().split(/\s+/);
    if(p[0]==='v')P.push([-(+p[1]),+p[2],+p[3]]);
    else if(p[0]==='vt')UV.push([+p[1],+p[2]]);
    else if(p[0]==='f'){
      for(let i=1;i<=3;i++){
        const t=p[i].split('/');const vi=+t[0]-1,ti=+t[1]-1,ni=+t[2]-1;
        const key=vi+'/'+ti+'/'+ni;
        if(idxMap[key]==null){
          idxMap[key]=pos.length/3;
          pos.push(P[vi][0],P[vi][1],P[vi][2]);
          if(UV[ti])uv.push(Math.min(0.998,Math.max(0.002,UV[ti][0])),Math.min(0.998,Math.max(0.002,UV[ti][1])));else uv.push(0.5,0.5);
          for(let k=0;k<4;k++){si.push(skin.skinIndex[vi*4+k]);sw.push(skin.skinWeight[vi*4+k]);}
        }
        outIdx.push(idxMap[key]);
      }
    }
  });
  let vol=0;
  for(let i=0;i<outIdx.length;i+=3){
    const a=outIdx[i]*3,b=outIdx[i+1]*3,c=outIdx[i+2]*3;
    vol+=(pos[a]*(pos[b+1]*pos[c+2]-pos[b+2]*pos[c+1])-pos[a+1]*(pos[b]*pos[c+2]-pos[b+2]*pos[c])+pos[a+2]*(pos[b]*pos[c+1]-pos[b+1]*pos[c]))/6;
  }
  if(vol<0)for(let i=0;i<outIdx.length;i+=3){const t=outIdx[i+1];outIdx[i+1]=outIdx[i+2];outIdx[i+2]=t;}
  const g=new THREE.BufferGeometry();
  g.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));
  g.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
  g.setAttribute('skinIndex',new THREE.Float32BufferAttribute(si,4));
  g.setAttribute('skinWeight',new THREE.Float32BufferAttribute(sw,4));
  g.setIndex(outIdx);
  g.computeVertexNormals();
  return g;
}

--- parseOBJStatic ---
function parseOBJStatic(txt){
  const P=[],UV=[],NO=[],idxMap={};
  const pos=[],uv=[],nor=[],outIdx=[];
  txt.split(/\r?\n/).forEach(line=>{
    const p=line.trim().split(/\s+/);
    if(p[0]==='v')P.push([+p[1],+p[2],+p[3]]);
    else if(p[0]==='vt')UV.push([+p[1],+p[2]]);
    else if(p[0]==='vn')NO.push([+p[1],+p[2],+p[3]]);
    else if(p[0]==='f'){
      for(let i=1;i<p.length;i++){
        const t=p[i].split('/');
        const vi=parseInt(t[0],10)-1,ti=parseInt(t[1],10)-1,ni=parseInt(t[2],10)-1;
        if(!(vi>=0&&P[vi]))continue;
        const key=vi+'/'+ti+'/'+ni;
        if(idxMap[key]==null){
          idxMap[key]=pos.length/3;
          pos.push(P[vi][0],P[vi][1],P[vi][2]);
          if(UV[ti])uv.push(UV[ti][0],UV[ti][1]);else uv.push(0,0);
          if(NO[ni])nor.push(NO[ni][0],NO[ni][1],NO[ni][2]);else nor.push(0,1,0);
        }
        outIdx.push(idxMap[key]);
      }
    }
  });
  const g=new THREE.BufferGeometry();
  g.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));
  g.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
  g.setAttribute('normal',new THREE.Float32BufferAttribute(nor,3));
  g.setIndex(outIdx);
  return g;
}

--- makeSkinnedModel ---
function makeSkinnedModel(geometry,data,texture,options={}){
  const model=new THREE.Group();
  const bones=data.bones.map(b=>{
    const bone=new THREE.Bone();
    bone.name=b.name;
    bone.position.fromArray(b.pos);
    bone.quaternion.fromArray(b.quat);
    if(b.scale)bone.scale.fromArray(b.scale);
    return bone;
  });
  data.bones.forEach((b,i)=>{if(b.parent>=0)bones[b.parent].add(bones[i]);});
  const inverses=validatedRigBoneInverses(data,bones);
  const skeleton=new THREE.Skeleton(data.skinBones.map(i=>bones[i]),inverses);
  if(texture)texture.colorSpace=THREE.SRGBColorSpace;
  const mesh=new THREE.SkinnedMesh(geometry,new THREE.MeshStandardMaterial({
    map:texture,roughness:0.85,metalness:0.02,transparent:false,opacity:1,depthWrite:true,alphaTest:0.5,
    side:THREE.FrontSide
  }));
  mesh.add(bones[0]);
  mesh.bind(skeleton,new THREE.Matrix4());
  mesh.frustumCulled=false;
  mesh.castShadow=true;
  model.add(mesh);
  const parents=data.bones.map(b=>b.parent);
  const bindLocal=data.bones.map(b=>new THREE.Quaternion(b.quat[0],b.quat[1],b.quat[2],b.quat[3]));
  const bindWorldQ=worldRotationsFromLocal(bindLocal,parents);
  const baseRestQ=options.baseRestQ||null;
  const sourceParents=baseRestQ?(options.baseParents||parents):parents;
  model.userData={
    bones,bindLocal,
    curQ:data.bones.map(b=>new THREE.Quaternion(b.quat[0],b.quat[1],b.quat[2],b.quat[3])),
    bindPos:data.bones.map(b=>new THREE.Vector3(b.pos[0],b.pos[1],b.pos[2])),
    curDelta:new THREE.Vector3(),mesh3:mesh,meshy:!!options.meshy,
    animatable:!!options.animatable,idleClipKey:options.idleClipKey||'idle2h',idleRole:options.idleRole,
    baseRestQ,baseRestWorldQ:baseRestQ?worldRotationsFromLocal(baseRestQ,sourceParents):null,
    bindWorldQ,parents,sourceParents,
    sourcePoseQ:data.bones.map(()=>new THREE.Quaternion()),
    sourceWorldQ:data.bones.map(()=>new THREE.Quaternion()),
    targetWorldQ:data.bones.map(()=>new THREE.Quaternion()),
    targetLocalQ:data.bones.map(()=>new THREE.Quaternion()),
    isClassRig:!!baseRestQ,grip:!!options.grip,role:options.role||'',gun:null,
    byName:Object.fromEntries(bones.map((bone,index)=>[bone.name,index]))
  };
  return model;
}

      if(gun.userData.flash){const cat=(WEAPONS[un.weapon]||{}).cat||'Fusiles de asalto';gun.userData.flash.position.x=(WEAPON_LEN[cat]||0.6)*0.96;gun.userData.flash.position.y=0;}
    }catch(e){}
  };
  const holder=makeRealGun3(un.weapon);
  if(holder)apply(holder);
  else{
    if(!gunSwapWaiters[un.weapon])gunSwapWaiters[un.weapon]=[];
    gunSwapWaiters[un.weapon].push(apply);
  }
}

function fitGunToHand3(un,grp,isLive){
  if(!HAS_THREE)return;
  const ud=grp&&grp.userData;if(!ud||!ud.gun||!ud.byName||!ud.bones)return;
  const parent=ud.gun.parent;if(!parent)return;
  if(grp.updateWorldMatrix)grp.updateWorldMatrix(true,true);else grp.updateMatrixWorld(true);
  const bR=ud.bones[ud.byName['Bip001 R Hand']];
  const bFA=ud.bones[ud.byName['Bip001 R Forearm']];
  if(!bR||!bFA)return;
  /* Socket anat├│mico del rig IPA: la mano derecha. Los metadatos de
     Barracks solo se usan en la armer├¡a; aqu├¡ manda la calibraci├│n real. */
  const gunFile=ud.gun.userData.file||(un&&WEAPON_MESH_FILE[un.weapon]);
  const cal=ud.gun.userData.calibration||weaponCalibration(gunFile);
  const socket=cal&&cal.socket?cal.socket:'Bip001 R Hand';
  const bS=socket&&ud.byName[socket]!==undefined?ud.bones[ud.byName[socket]]:null;
  if(bS){
    _gpRPalm.setFromMatrixPosition(bS.matrixWorld);
    parent.worldToLocal(_gpRPalm);
    const off=cal&&cal.gripOffset?cal.gripOffset:[0,0,0];
    ud.gun.position.set(_gpRPalm.x+off[0],_gpRPalm.y+off[1],_gpRPalm.z+off[2]);
    bS.getWorldQuaternion(_gpSocketQ);
    parent.getWorldQuaternion(_gpParentQ);
    ud.gun.quaternion.copy(_gpParentQ.invert().multiply(_gpSocketQ));
    return;
  }
  _gpRPalm.copy(_gpPalmR).applyMatrix4(bR.matrixWorld);
  parent.worldToLocal(_gpRPalm);
  _gpFA.setFromMatrixPosition(bFA.matrixWorld);
  parent.worldToLocal(_gpFA);
  const bL=ud.bones[ud.byName['Bip001 L Hand']];
  let hasL=false;
  if(bL){
    _gpLPalm.copy(_gpPalmL).applyMatrix4(bL.matrixWorld);
    parent.worldToLocal(_gpLPalm);
    hasL=true;
  }
  const lk=roleLook(un.role);
  const ck=ud._clipKey||'';
  const is1h=(ud.gun.userData.cat==='Pistolas');
  const force1h=(ck==='death'||ck==='deathExpl'||ck==='hit'||ck==='hitBurst'||ck==='hitExpl'||ck==='throw'||ck==='deploy'||ck==='interact'||ck==='victory'||ck==='pickup');
  const pose=gunPoseFromHand(_gpRPalm,_gpFA,ud.gun.userData.grip,hasL?_gpLPalm:null,{
    is1h:is1h,
    force1h:force1h,
    roleH:lk.s[1],
    roleAvg:(lk.s[0]+lk.s[1]+lk.s[2])/3
  });
  if(!pose)return;
  if(isLive&&ud._firePulse>0){
    pose.origin.addScaledVector(pose.dir,-0.055*ud._firePulse);
    pose.quat.multiply(_gpRecoil.setFromAxisAngle(_gpUp,0.035*ud._firePulse));
  }
  ud.gun.position.copy(pose.origin);
  ud.gun.quaternion.copy(pose.quat);
}

--- mount ---
function mount(model){
    while(viewer.spin.children.length)viewer.spin.remove(viewer.spin.children[0]);
    if(model.userData&&model.userData.bones&&model.userData.bindLocal&&CLIPLIB.ready)
      poseViewerRest(model,model.userData.idleClipKey||'idle2h',0);
    model.updateMatrixWorld(true);
    model.traverse(object=>{
      if(object.isSkinnedMesh&&object.skeleton)object.skeleton.update();
    });
    model.updateMatrixWorld(true);
    const bounds=new THREE.Box3();
    const meshBounds=new THREE.Box3();
    model.traverse(object=>{
      if(!object.isMesh||!object.geometry)return;
      /* El arma equipada no entra en el encuadre: si contara, el ca├▒├│n
         desplazar├¡a el centro y el soldado quedar├¡a descentrado. */
      let p=object.parent;
      while(p&&p!==model){if(p.userData&&p.userData.isViewerGun)return;p=p.parent;}
      if(object.isSkinnedMesh){
        if(object.computeBoundingBox)object.computeBoundingBox();
        if(object.boundingBox&&!object.boundingBox.isEmpty()){
          meshBounds.copy(object.boundingBox).applyMatrix4(object.matrixWorld);
        }else meshBounds.setFromObject(object,true);
      }else meshBounds.setFromObject(object,true);
      bounds.union(meshBounds);
    });
    const size=new THREE.Vector3(),center=new THREE.Vector3();
    bounds.getSize(size);bounds.getCenter(center);
    if(!Number.isFinite(size.x)||Math.max(size.x,size.y,size.z)<0.001)throw new Error('El modelo no contiene geometr├¡a visible.');
    const centerOffset=viewerCenterOffset(center);
    const centered=new THREE.Group();
    centered.position.set(centerOffset.x,centerOffset.y,centerOffset.z);
    centered.add(model);
    viewer.spin.add(centered);
    centered.updateMatrixWorld(true);
    viewer.idleActor=viewer.kind==='roles'&&model.userData.animatable?model:null;
    viewer.gripActor=viewer.kind==='roles'&&model.userData.grip?model:null;
    if(viewer.meshyFill)viewer.meshyFill.intensity=viewer.kind==='roles'&&model.userData.meshy?1.8:0;
    if(viewer.meshyRim)viewer.meshyRim.intensity=viewer.kind==='roles'&&model.userData.meshy?2.2:0;
    viewer.idleClipKey=model.userData.idleClipKey||'idle2h';
    if(viewer.gripActor&&viewer.gripActor.userData.gun)
      fitGunToHand3(viewer.gripActor.userData.gripUnit,viewer.gripActor,false);
    viewer.spin.rotation.set(viewer.kind==='weapons'?0.25:0,-0.55,0);
    const platformRadius=Math.max(0.35,Math.max(size.x,size.z)*0.72);
    viewer.platform.scale.set(platformRadius,platformRadius,1);
    viewer.platform.position.y=-size.y/2-0.035;
    resize();
    viewer.baseDistance=fitViewerDistance(size,viewer.camera.fov,viewer.camera.aspect);
    updateCamera();
    viewer.renderer.render(viewer.scene,viewer.camera);
    refreshTexturePreview();
    setStatus('',false);
  }

async function makeViewerOriginal(entry,skin){
    /* La geometr├¡a la decide el cuerpo (entry.body), no el rol: cada
       personaje muestra la malla de su cuerpo. Los 4 roles IPA usan
       REAL.byRole; los cuerpos variantes se cargan de bodies/; el
       cuerpo Soldier_Light usa el soldado gen├®rico (REAL). */
    let source=null,bodyGeo=null;
    if(entry.bodyFile){
      bodyGeo=await loadViewerBody(entry.bodyFile);
    }else if(entry.bodyRole){
      source=REAL.byRole[entry.bodyRole];
      if(!source)throw new Error('El modelo de clase '+entry.bodyRole+' todav├¡a no est├í disponible.');
    }else{
      source=REAL;
    }
    let texture=null;
    if(skin.key&&skin.key.indexOf('cast:')===0){
      const texturePath='assets/soldier/cast/'+skin.key.slice(5);
      if(viewer.textures[texturePath])texture=viewer.textures[texturePath];
      else{
        texture=await new Promise((resolve,reject)=>{
          new THREE.TextureLoader().load(texturePath,resolve,undefined,reject);
        });
        texture.colorSpace=THREE.SRGBColorSpace;
        viewer.textures[texturePath]=texture;
      }
    }else if(skin.key==='native'&&source){
      texture=source.tex;
    }
    if(!texture)throw new Error('El modelo no tiene una textura disponible.');
    /* Solo el cuerpo Soldier_Light del repo (Hipster Dave, el soldado
       gen├®rico) usa el modelo con esqueleto y pose de reposo animada.
       Sister Florence se monta tambi├®n con esqueleto, pero con la pose
       de reposo retargetizada y sin animaci├│n, para que el arma se
       enganche a sus manos. El resto sigue siendo malla est├ítica:
       forzar animatable en todos es lo que descabal├│ los modelos. */
    if(!entry.bodyFile&&!entry.bodyRole){
      const model=makeViewerRiggedModel(source.geo.clone(),source.data,texture,{animatable:true,grip:true});
      model.scale.setScalar(1.15);
      const mesh=model.userData.mesh3;
      model.remove(mesh);
      const spin=new THREE.Group();
      spin.rotation.y=Math.PI/2;
      model.add(spin);
      const tilt=new THREE.Group();
      tilt.rotation.x=-Math.PI/2;
      spin.add(tilt);
      mesh.scale.setScalar(0.95);
      tilt.add(mesh);
      return model;
    }
    if(entry.name==='Sister Florence'&&source){
      const started=Date.now();
      while(!CLIPLIB.ready&&Date.now()-started<5000)await wait(50);
      const baseData=sameRigBoneOrder(REAL.data&&REAL.data.bones,source.data.bones)?REAL.data:null;
      const baseRestQ=baseData?baseData.bones.map(b=>new THREE.Quaternion(b.quat[0],b.quat[1],b.quat[2],b.quat[3])):null;
      const baseParents=baseData?baseData.bones.map(b=>b.parent):null;
      const model=makeViewerRiggedModel(source.geo.clone(),source.data,texture,{
        baseRestQ,baseParents,grip:true,role:entry.role
      });
      model.scale.setScalar(1.15);
      const mesh=model.userData.mesh3;
      model.remove(mesh);
      const spin=new THREE.Group();
      spin.rotation.y=Math.PI/2;
      const lk=roleLook(entry.role);
      if(lk&&lk.s)spin.scale.set(lk.s[0],lk.s[1],lk.s[2]);
      model.add(spin);
      const tilt=new THREE.Group();
      tilt.rotation.x=-Math.PI/2;
      spin.add(tilt);
      mesh.scale.setScalar(0.95);
      tilt.add(mesh);
      poseViewerRest(model,'idle2h',0);
      model.userData.gripUnit={role:entry.role,weapon:''};
      return model;
    }
    const geometry=bodyGeo?bodyGeo.clone():(source?source.geo.clone():null);
    if(!geometry)throw new Error('El modelo no tiene una geometr├¡a disponible.');
    geometry.deleteAttribute('skinIndex');
    geometry.deleteAttribute('skinWeight');
    geometry.computeVertexNormals();
    const mesh=new THREE.Mesh(geometry,new THREE.MeshStandardMaterial({
      map:texture,roughness:0.85,metalness:0.02,transparent:false,opacity:1,depthWrite:true,
      side:THREE.DoubleSide
    }));
    mesh.castShadow=true;
    const model=new THREE.Group();
    model.scale.setScalar(1.15);
    const spin=new THREE.Group();spin.rotation.y=Math.PI/2;model.add(spin);
    const tilt=new THREE.Group();tilt.rotation.x=-Math.PI/2;spin.add(tilt);
    mesh.scale.setScalar(0.95);tilt.add(mesh);
    model.userData.staticRole=true;
    return model;
  }

--- loadViewerMeshy ---
async function loadViewerMeshy(entry){
    const base='assets/personajes_propios/'+entry.mesh;
    const [response,rigResponse,texture]=await Promise.all([
      fetch(base+'.obj'),fetch(base+'_rig.json'),
      new Promise((resolve,reject)=>{
        /* Cada dise├▒o Meshy es autocontenido en assets/personajes_propios/
           (obj + rig + tex): la textura ya no es la de clase en
           assets/soldier/, porque esas clases volvieron a ser del original. */
        new THREE.TextureLoader().load(base+'_tex.png',resolve,undefined,reject);
      })
    ]);
    if(!response.ok)throw new Error('No se pudo cargar el OBJ Meshy de '+entry.mesh+'.');
    if(!rigResponse.ok)throw new Error('No se pudo cargar el rig Meshy de '+entry.mesh+'.');
    texture.colorSpace=THREE.SRGBColorSpace;
    const [objText,rawRig]=await Promise.all([response.text(),rigResponse.json()]);
    const rig=normalizeMeshyRig(rawRig);
    const geometry=parseViewerOBJ(objText,rig);
    const started=Date.now();
    while(!CLIPLIB.ready&&Date.now()-started<5000)await wait(50);
    const baseData=sameRigBoneOrder(REAL.data&&REAL.data.bones,rig.bones)?REAL.data:null;
    const baseRestQ=baseData?baseData.bones.map(b=>new THREE.Quaternion(b.quat[0],b.quat[1],b.quat[2],b.quat[3])):null;
    const baseParents=baseData?baseData.bones.map(b=>b.parent):null;
    const model=makeViewerRiggedModel(geometry,rig,texture,{
      meshy:true,grip:true,role:entry.role,baseRestQ,baseParents
    });
    if(rig.bones[1].pos[2]>rig.bones[1].pos[1]+0.1)model.rotation.x=-Math.PI/2;
    poseViewerRest(model,'idle2h',0);
    model.userData.gripUnit={role:entry.role,weapon:''};
    return model;
  }

async function syncViewerGun(model,entry,token){
    const wanted=viewer.weapon||'';
    const current=(model.userData&&model.userData.viewerGunWeapon)||'';
    if(current===wanted)return;
    if(model.userData&&model.userData.viewerGun){
      model.remove(model.userData.viewerGun);
      if(model.userData.gun===model.userData.viewerGun)model.userData.gun=null;
      model.userData.viewerGun=null;
      model.userData.viewerGunWeapon='';
    }
    if(!wanted)return;
    const meshFile=WEAPON_MESH_FILE[wanted];
    if(!meshFile)return;
    const cacheKey=viewerWeaponCacheKey(meshFile);
    if(!GUN_MESH_CACHE[cacheKey]){
      const started=Date.now();
      while(GUN_MESH_CACHE[cacheKey]===null&&Date.now()-started<10000){
        if(token!==viewer.selection)return;
        await wait(100);
      }
      if(GUN_MESH_CACHE[cacheKey]===false)throw new Error('No se pudo cargar la malla de esta arma.');
      if(!GUN_MESH_CACHE[cacheKey])throw new Error('La carga de la malla tard├│ demasiado.');
    }
    const holder=GUN_MESH_CACHE[cacheKey].clone(true);
    const gunMeta=holder.userData.meta||weaponMeta(meshFile);
    let gunGroup;
    if(model.userData&&model.userData.grip){
      gunGroup=makeGunProp3({role:entry.role,weapon:wanted});
      gunGroup.userData.isViewerGun=true;
      gunGroup.children.forEach(ch=>{ch.visible=false;});
      gunGroup.add(holder);
    }else{
      gunGroup=new THREE.Group();
      gunGroup.userData.isViewerGun=true;
      if(entry.mesh){
        const front=VIEWER_MESHY_FRONT[entry.mesh]||1;
        gunGroup.position.set(0,0.62,front*0.14);
        gunGroup.rotation.y=front>0?-Math.PI/2:Math.PI/2;
      }else{
        gunGroup.position.set(0.14,0.62,-0.02);
      }
      gunGroup.add(holder);
    }
    gunGroup.userData.file=meshFile;
    gunGroup.userData.meta=gunMeta;
    if(DEBUG_WEAPONS){
      const marker=new THREE.Mesh(new THREE.SphereGeometry(0.045,12,8),new THREE.MeshBasicMaterial({color:0xff3300,depthTest:false}));
      marker.userData.isViewerGun=true;
      gunGroup.add(marker);
      console.info('[debugWeapons]',meshFile,gunMeta&&gunMeta.attachmentJoint,holder.userData.calibration);
    }
    model.add(gunGroup);
    model.userData.viewerGun=gunGroup;
    model.userData.viewerGunWeapon=wanted;
    if(model.userData.grip){
      model.userData.gun=gunGroup;
      model.userData.gripUnit={role:entry.role,weapon:wanted};
      fitGunToHand3(model.userData.gripUnit,model,false);
    }
  }

--- poseViewerRest ---
function poseViewerRest(model,clipKey,ct){
    const pack=CLIPLIB.pack,clip=pack&&pack.clips&&pack.clips[clipKey];
    const ud=model&&model.userData;
    if(!clip||!ud||!ud.bones||!ud.bindLocal)return false;
    const tq=new THREE.Quaternion();
    const retarget=!!(ud.isClassRig&&ud.baseRestWorldQ&&ud.sourcePoseQ);
    const rootP=new THREE.Vector3(),rootT=new THREE.Vector3(),pelvis=new THREE.Vector3();
    for(let n=0;n<24;n++){
      for(let i=0;i<ud.bones.length;i++){
        const tr=clip.rot[pack.bones[i]];
        const restQ=retarget?ud.baseRestQ[i]:ud.bindLocal[i];
        if(tr)sampleClipQuat(tr,ct,tq);else tq.copy(restQ);
        if(retarget){
          ud.sourcePoseQ[i].copy(tq);
        }else{
          ud.bones[i].quaternion.copy(tq);
        }
      }
      if(retarget){
        retargetPoseToRig(ud.sourcePoseQ,ud.baseRestWorldQ,ud.bindWorldQ,ud.sourceParents,ud.parents,ud.sourceWorldQ,ud.targetWorldQ,ud.targetLocalQ);
        for(let i=0;i<ud.bones.length;i++)ud.bones[i].quaternion.copy(ud.targetLocalQ[i]);
      }
      if(clip.root)sampleClipPos(clip.root,ct,rootT);else rootT.set(0,0,0);
      rootP.copy(ud.curDelta).lerp(rootT,0.45);
      pelvis.copy(ud.bindPos[1]);
      rootBonePositionForClip(ud.bindPos[0],pelvis,rootP,rootT.copy(pelvis).applyQuaternion(ud.bones[0].quaternion),ud.bones[0].position);
    }
    ud._clipKey=clipKey;
    model.updateMatrixWorld(true);
    return true;
  }
