function makeRealGun3(weapon,preservePivot){
  const file=WEAPON_MESH_FILE[weapon]; if(!file)return null;
  const cached=GUN_MESH_CACHE[weapon]; if(cached!==undefined)return cached;
  GUN_MESH_CACHE[weapon]=null; /* cargando */
  fetch('assets/weapons/'+file+'.obj').then(r=>{if(!r.ok)throw new Error('obj');return r.text();}).then(txt=>{
    const geo=parseOBJStatic(txt);
    geo.computeBoundingBox();
    const bb=geo.boundingBox, size=new THREE.Vector3(); bb.getSize(size);
    const long=Math.max(size.x,size.z,0.001);
    const cat=(WEAPONS[weapon]||{}).cat||'Fusiles de asalto';
    const target=WEAPON_LEN[cat]||0.6;
    const k=target/long;
    const meta=weaponMeta(file);
    const mat=new THREE.MeshStandardMaterial({color:0x9a9a9a,roughness:0.72,metalness:0.25});
    const texLoad=new THREE.TextureLoader();
    texLoad.load('assets/weapons/'+file+'_tex.png',t=>{t.colorSpace=THREE.SRGBColorSpace;mat.map=t;mat.color.set(0xffffff);mat.needsUpdate=true;},undefined,()=>{});
    const mesh=new THREE.Mesh(geo,mat);
    /* malla en Z arriba -> Y arriba */
    const c=new THREE.Vector3(); bb.getCenter(c);
    const pivot=[-bb.min.x,-c.z,-c.y];
    const holder=new THREE.Group();
    holder.add(mesh);
    if(preservePivot){
      /* El .obj conserva el pivote del prefab: no se recentra. Se usa el
         offset/scale de UnitWeapon y el socket del esqueleto. */
      const off=meta&&meta.barracksOffset?meta.barracksOffset:[0,0,0];
      mesh.position.set(0,0,0);
      holder.position.set(off[0],off[1],off[2]);
      holder.scale.setScalar(meta&&meta.barracksScale?meta.barracksScale[0]:k);
    }else{
      /* La culata queda en el origen: el ca├▒├│n sale hacia +X. */
      mesh.position.set(pivot[0],pivot[1],pivot[2]);
      holder.scale.setScalar(k);
    }
    holder.rotation.x=-Math.PI/2;
    holder.userData={file:file,meta:meta||null,pivot:pivot};
    GUN_MESH_CACHE[weapon]=holder;
    if(gunSwapWaiters[weapon]){gunSwapWaiters[weapon].forEach(fn=>{try{fn(holder);}catch(e){}});gunSwapWaiters[weapon]=[];}
  }).catch(()=>{delete GUN_MESH_CACHE[weapon];});
  return null;
}

function fitGunToHand3(un,grp,isLive){
  if(!HAS_THREE)return;
  const ud=grp&&grp.userData;if(!ud||!ud.gun||!ud.byName||!ud.bones)return;
  const parent=ud.gun.parent;if(!parent)return;
  if(grp.updateWorldMatrix)grp.updateWorldMatrix(true,true);else grp.updateMatrixWorld(true);
  const bR=ud.bones[ud.byName['Bip001 R Hand']];
  const bFA=ud.bones[ud.byName['Bip001 R Forearm']];
  if(!bR||!bFA)return;
  /* Socket real de UnitWeapon: se ancla al hueso gunBone_R del esqueleto
     (aliasado a la mano derecha en los rigs del repo) y se aplica el offset
     de Barracks. Si el hueso o los metadatos no est├ín, sigue el ajuste
     manual por palmas de abajo. */
  const gunFile=ud.gun.userData.file||(un&&WEAPON_MESH_FILE[un.weapon]);
  const meta=ud.gun.userData.meta||weaponMeta(gunFile);
  const socket=meta&&meta.attachmentJoint;
  const bS=socket&&ud.byName[socket]!==undefined?ud.bones[ud.byName[socket]]:null;
  if(bS){
    _gpRPalm.setFromMatrixPosition(bS.matrixWorld);
    parent.worldToLocal(_gpRPalm);
    const off=meta.barracksOffset||[0,0,0];
    ud.gun.position.set(_gpRPalm.x+off[0],_gpRPalm.y+off[1],_gpRPalm.z+off[2]);
    bS.getWorldQuaternion(_gpSocketQ);
    parent.getWorldQuaternion(_gpParentQ);
    ud.gun.quaternion.copy(_gpParentQ.invert().multiply(_gpSocketQ));
    return;
  }
  _gpRPalm.copy(_gpPalmR).applyMatrix4(bR.matrixWorld);
  parent.worldToLocal(_gpRPalm);
  _gpFA.setFromMatrixPosition(bFA.matrixWorld);
  parent.worldToLocal(_gpFA);
  const bL=ud.bones[ud.byName['Bip001 L Hand']];
  let hasL=false;
  if(bL){
    _gpLPalm.copy(_gpPalmL).applyMatrix4(bL.matrixWorld);
    parent.worldToLocal(_gpLPalm);
    hasL=true;
  }
  const lk=roleLook(un.role);
  const ck=ud._clipKey||'';
  const is1h=(ud.gun.userData.cat==='Pistolas');
  const force1h=(ck==='death'||ck==='deathExpl'||ck==='hit'||ck==='hitBurst'||ck==='hitExpl'||ck==='throw'||ck==='deploy'||ck==='interact'||ck==='victory'||ck==='pickup');
  const pose=gunPoseFromHand(_gpRPalm,_gpFA,ud.gun.userData.grip,hasL?_gpLPalm:null,{
    is1h:is1h,
    force1h:force1h,
    roleH:lk.s[1],
    roleAvg:(lk.s[0]+lk.s[1]+lk.s[2])/3
  });
  if(!pose)return;
  if(isLive&&ud._firePulse>0){
    pose.origin.addScaledVector(pose.dir,-0.055*ud._firePulse);
    pose.quat.multiply(_gpRecoil.setFromAxisAngle(_gpUp,0.035*ud._firePulse));
  }
  ud.gun.position.copy(pose.origin);
  ud.gun.quaternion.copy(pose.quat);
}

--- gunPoseFromHand ---
function gunPoseFromHand(rPalm,fa,grip,lPalm,opts){
  _gpDir.copy(rPalm).sub(fa);
  if(_gpDir.lengthSq()<1e-6)return null;
  _gpDir.normalize();
  const is1h=!!(opts&&opts.is1h);
  const roleH=(opts&&opts.roleH)||1;
  const roleAvg=(opts&&opts.roleAvg)||1;
  _gpDir.lerp(_gpFwd,is1h?0.30:0.32).normalize();
  let w2h=0;
  if(!is1h&&lPalm&&!(opts&&opts.force1h)){
    const span=rPalm.distanceTo(lPalm)/roleAvg;
    const ly=lPalm.y/roleH;
    w2h=_smooth01(0.28,0.42,ly)*_smooth01(0.20,0.28,span)*(1-_smooth01(0.52,0.62,span));
    if(w2h>0.001){
      _gpTwoH.copy(lPalm).sub(rPalm);
      if(_gpTwoH.lengthSq()>1e-6){
        _gpTwoH.normalize();
        _gpDir.lerp(_gpTwoH,w2h).normalize();
      }
    }
  }
  _gpZ.crossVectors(_gpDir,_gpUp);
  if(_gpZ.lengthSq()<1e-4)_gpZ.set(0,0,1);else _gpZ.normalize();
  _gpY.crossVectors(_gpZ,_gpDir).normalize();
  const g=grip||{x:0.2,y:-0.05,scale:1};
  const gs=g.scale||1;
  _gpOrigin.copy(rPalm).addScaledVector(_gpDir,-g.x*gs).addScaledVector(_gpY,-g.y*gs);
  _gpMat.makeBasis(_gpDir,_gpY,_gpZ);
  _gpQuat.setFromRotationMatrix(_gpMat);
  return {origin:_gpOrigin,dir:_gpDir,quat:_gpQuat};
}

--- retargetPoseToRig ---
function retargetPoseToRig(sourcePoseQ,sourceRestWorldQ,targetRestWorldQ,sourceParents,targetParents,sourceWorldQ,targetWorldQ,targetLocalQ){
  for(let i=0;i<sourcePoseQ.length;i++){
    const sourceParent=sourceParents[i];
    if(sourceParent>=0)sourceWorldQ[i].copy(sourceWorldQ[sourceParent]).multiply(sourcePoseQ[i]);
    else sourceWorldQ[i].copy(sourcePoseQ[i]);
    _retargetDelta.copy(sourceWorldQ[i])
      .multiply(_retargetInverse.copy(sourceRestWorldQ[i]).invert())
      .multiply(targetRestWorldQ[i]);
    targetWorldQ[i].copy(_retargetDelta);
    const targetParent=targetParents[i];
    if(targetParent>=0)targetLocalQ[i].copy(targetWorldQ[targetParent]).invert().multiply(targetWorldQ[i]);
    else targetLocalQ[i].copy(targetWorldQ[i]);
  }
}

function rootBonePositionForClip(rootBase,pelvisOffset,delta,rotatedPelvis,out){
  out.x=rootBase.x+pelvisOffset.x+delta.x-rotatedPelvis.x;
  out.y=rootBase.y+pelvisOffset.y+delta.y-rotatedPelvis.y;
  out.z=rootBase.z+pelvisOffset.z+delta.z-rotatedPelvis.z;
  return out;
}

function poseSpriteClip(un,grp,clip,ct,clipKey){
  const pack=CLIPLIB.pack;if(!pack||!clip)return false;
  const ud=grp.userData;if(!ud||!ud.bones||!ud.bindLocal||!ud.curQ)return false;
  ud._clipKey=clipKey||'';
  const tq=new THREE.Quaternion(),tp=new THREE.Vector3(),tp2=new THREE.Vector3(),pelvis=new THREE.Vector3();
  const retarget=!!(ud.isClassRig&&ud.baseRestWorldQ&&ud.sourcePoseQ);
  for(let n=0;n<24;n++){
    for(let i=0;i<ud.bones.length;i++){
      const tr=clip.rot[pack.bones[i]];
      const restQ=retarget?ud.baseRestQ[i]:ud.bindLocal[i];
      if(tr)sampleClipQuat(tr,ct,tq);else tq.copy(restQ);
      if(retarget){
        ud.sourcePoseQ[i].copy(tq);
      }else{
        ud.curQ[i].slerp(tq,0.45);
        ud.bones[i].quaternion.copy(ud.curQ[i]);
      }
    }
    if(retarget){
      retargetPoseToRig(ud.sourcePoseQ,ud.baseRestWorldQ,ud.bindWorldQ,ud.sourceParents,ud.parents,ud.sourceWorldQ,ud.targetWorldQ,ud.targetLocalQ);
      for(let i=0;i<ud.bones.length;i++){
        ud.curQ[i].slerp(ud.targetLocalQ[i],0.45);
        ud.bones[i].quaternion.copy(ud.curQ[i]);
      }
    }
    if(clip.root)sampleClipPos(clip.root,ct,tp);else tp.set(0,0,0);
    ud.curDelta.lerp(tp,0.45);
    pelvis.copy(ud.bindPos[1]);
    rootBonePositionForClip(ud.bindPos[0],pelvis,ud.curDelta,tp2.copy(pelvis).applyQuaternion(ud.bones[0].quaternion),ud.bones[0].position);
  }
  return true;
}

function mount(model){
    while(viewer.spin.children.length)viewer.spin.remove(viewer.spin.children[0]);
    model.updateMatrixWorld(true);
    model.traverse(object=>{
      if(object.isSkinnedMesh&&object.skeleton)object.skeleton.update();
    });
    model.updateMatrixWorld(true);
    const bounds=new THREE.Box3();
    const meshBounds=new THREE.Box3();
    model.traverse(object=>{
      if(!object.isMesh||!object.geometry)return;

    const centerOffset=viewerCenterOffset(center);
    const centered=new THREE.Group();

    viewer.spin.add(centered);

    viewer.gripActor=viewer.kind==='roles'&&model.userData.grip?model:null;

async function loadViewerBody(file){
    const path='assets/soldier/bodies/'+file;
    if(viewer.bodyGeo[path])return viewer.bodyGeo[path];
    const text=await fetch(path).then(r=>{if(!r.ok)throw new Error('No se pudo cargar el cuerpo '+file);return r.text();});
    let count=0;
    for(const line of text.split('\n')){if(line.startsWith('v '))count++;}
    const skin={skinIndex:new Array(count*4).fill(0),skinWeight:new Array(count*4).fill(0)};
    const geo=parseOBJ3(text,skin);
    const pos=geo.attributes.position.array;
    const zs=[];
    for(let i=2;i<pos.length;i+=3)zs.push(pos[i]);
    zs.sort((a,b)=>a-b);
    const p95=zs[Math.floor(zs.length*0.95)]||1;
    const scale=1.14/p95;
    if(scale!==1)geo.scale(scale,scale,scale);
    viewer.bodyGeo[path]=geo;
    return geo;
  }

async function makeViewerOriginal(entry,skin){
    /* La geometr├¡a la decide el cuerpo (entry.body), no el rol: cada
       personaje muestra la malla de su cuerpo. Los 4 roles IPA usan
       REAL.byRole; los cuerpos variantes se cargan de bodies/; el
       cuerpo Soldier_Light usa el soldado gen├®rico (REAL). */
    let source=null,bodyGeo=null;
    if(entry.bodyFile){
      bodyGeo=await loadViewerBody(entry.bodyFile);
    }else if(entry.bodyRole){
      source=REAL.byRole[entry.bodyRole];
      if(!source)throw new Error('El modelo de clase '+entry.bodyRole+' todav├¡a no est├í disponible.');
    }else{
      source=REAL;
    }
    let texture=null;
    if(skin.key&&skin.key.indexOf('cast:')===0){
      const texturePath='assets/soldier/cast/'+skin.key.slice(5);
      if(viewer.textures[texturePath])texture=viewer.textures[texturePath];
      else{
        texture=await new Promise((resolve,reject)=>{
          new THREE.TextureLoader().load(texturePath,resolve,undefined,reject);
        });
        texture.colorSpace=THREE.SRGBColorSpace;
        viewer.textures[texturePath]=texture;
      }
    }else if(skin.key==='native'&&source){
      texture=source.tex;
    }
    if(!texture)throw new Error('El modelo no tiene una textura disponible.');
    /* Solo el cuerpo Soldier_Light del repo (Hipster Dave, el soldado
       gen├®rico) usa el modelo con esqueleto y pose de reposo animada.
       Sister Florence se monta tambi├®n con esqueleto, pero con la pose
       de reposo retargetizada y sin animaci├│n, para que el arma se
       enganche a sus manos. El resto sigue siendo malla est├ítica:
       forzar animatable en todos es lo que descabal├│ los modelos. */
    if(!entry.bodyFile&&!entry.bodyRole){
      const model=makeViewerRiggedModel(source.geo.clone(),source.data,texture,{animatable:true,grip:true});
      model.scale.setScalar(1.15);
      const mesh=model.userData.mesh3;
      model.remove(mesh);
      const spin=new THREE.Group();
      spin.rotation.y=Math.PI/2;
      model.add(spin);
      const tilt=new THREE.Group();
      tilt.rotation.x=-Math.PI/2;
      spin.add(tilt);
      mesh.scale.setScalar(0.95);
      tilt.add(mesh);
      return model;
    }
    if(entry.name==='Sister Florence'&&source){
      const started=Date.now();
      while(!CLIPLIB.ready&&Date.now()-started<5000)await wait(50);
      const baseData=sameRigBoneOrder(REAL.data&&REAL.data.bones,source.data.bones)?REAL.data:null;
      const baseRestQ=baseData?baseData.bones.map(b=>new THREE.Quaternion(b.quat[0],b.quat[1],b.quat[2],b.quat[3])):null;
      const baseParents=baseData?baseData.bones.map(b=>b.parent):null;
      const model=makeViewerRiggedModel(source.geo.clone(),source.data,texture,{
        baseRestQ,baseParents,grip:true,role:entry.role
      });
      model.scale.setScalar(1.15);
      const mesh=model.userData.mesh3;
      model.remove(mesh);
      const spin=new THREE.Group();
      spin.rotation.y=Math.PI/2;
      const lk=roleLook(entry.role);
      if(lk&&lk.s)spin.scale.set(lk.s[0],lk.s[1],lk.s[2]);
      model.add(spin);
      const tilt=new THREE.Group();
      tilt.rotation.x=-Math.PI/2;
      spin.add(tilt);
      mesh.scale.setScalar(0.95);
      tilt.add(mesh);
      poseViewerRest(model,'idle2h',0);
      model.userData.gripUnit={role:entry.role,weapon:''};
      return model;
    }
    const geometry=bodyGeo?bodyGeo.clone():(source?source.geo.clone():null);
    if(!geometry)throw new Error('El modelo no tiene una geometr├¡a disponible.');
    geometry.deleteAttribute('skinIndex');
    geometry.deleteAttribute('skinWeight');
    geometry.computeVertexNormals();
    const mesh=new THREE.Mesh(geometry,new THREE.MeshStandardMaterial({
      map:texture,roughness:0.85,metalness:0.02,transparent:false,opacity:1,depthWrite:true,
      side:THREE.DoubleSide
    }));
    mesh.castShadow=true;
    const model=new THREE.Group();
    model.scale.setScalar(1.15);
    const spin=new THREE.Group();spin.rotation.y=Math.PI/2;model.add(spin);
    const tilt=new THREE.Group();tilt.rotation.x=-Math.PI/2;spin.add(tilt);
    mesh.scale.setScalar(0.95);tilt.add(mesh);
    model.userData.staticRole=true;
    return model;
  }

async function syncViewerGun(model,entry,token){
    const wanted=viewer.weapon||'';
    const current=(model.userData&&model.userData.viewerGunWeapon)||'';
    if(current===wanted)return;
    if(model.userData&&model.userData.viewerGun){
      model.remove(model.userData.viewerGun);
      if(model.userData.gun===model.userData.viewerGun)model.userData.gun=null;
      model.userData.viewerGun=null;
      model.userData.viewerGunWeapon='';
    }
    if(!wanted)return;
    const meshFile=WEAPON_MESH_FILE[wanted];
    if(!meshFile)return;
    const cacheKey=viewerWeaponCacheKey(meshFile);
    if(!GUN_MESH_CACHE[cacheKey]){
      const started=Date.now();
      while(GUN_MESH_CACHE[cacheKey]===null&&Date.now()-started<10000){
        if(token!==viewer.selection)return;
        await wait(100);
      }
      if(GUN_MESH_CACHE[cacheKey]===false)throw new Error('No se pudo cargar la malla de esta arma.');
      if(!GUN_MESH_CACHE[cacheKey])throw new Error('La carga de la malla tard├│ demasiado.');
    }
    const holder=GUN_MESH_CACHE[cacheKey].clone(true);
    const gunMeta=holder.userData.meta||weaponMeta(meshFile);
    let gunGroup;
    if(model.userData&&model.userData.grip){
      gunGroup=makeGunProp3({role:entry.role,weapon:wanted});
      gunGroup.userData.isViewerGun=true;
      gunGroup.children.forEach(ch=>{ch.visible=false;});
      gunGroup.add(holder);
    }else{
      gunGroup=new THREE.Group();
      gunGroup.userData.isViewerGun=true;
      if(entry.mesh){
        const front=VIEWER_MESHY_FRONT[entry.mesh]||1;
        gunGroup.position.set(0,0.62,front*0.14);
        gunGroup.rotation.y=front>0?-Math.PI/2:Math.PI/2;
      }else{
        gunGroup.position.set(0.14,0.62,-0.02);
      }
      gunGroup.add(holder);
    }
    gunGroup.userData.file=meshFile;
    gunGroup.userData.meta=gunMeta;
    model.add(gunGroup);
    model.userData.viewerGun=gunGroup;
    model.userData.viewerGunWeapon=wanted;
    if(model.userData.grip){
      model.userData.gun=gunGroup;
      model.userData.gripUnit={role:entry.role,weapon:wanted};
      fitGunToHand3(model.userData.gripUnit,model,false);
    }
  }
