'use strict';
const test = require('node:test');
const assert = require('node:assert/strict');
const fs = require('node:fs');
const path = require('node:path');
const vm = require('node:vm');

const root = path.dirname(__dirname);
const html = fs.readFileSync(path.join(root, 'index.html'), 'utf8');

function extraerFn(src, nombre) {
  const inicio = src.indexOf('function ' + nombre + '(');
  assert.notEqual(inicio, -1, 'no existe ' + nombre);
  let i = src.indexOf('{', inicio);
  let depth = 0;
  let quote = null;
  let escape = false;
  for (; i < src.length; i++) {
    const c = src[i];
    if (quote) {
      if (escape) escape = false;
      else if (c === '\\') escape = true;
      else if (c === quote) quote = null;
      continue;
    }
    if (c === "'" || c === '"' || c === '`') { quote = c; continue; }
    if (c === '{') depth++;
    else if (c === '}') {
      depth--;
      if (depth === 0) return src.slice(inicio, i + 1);
    }
  }
  throw new Error('llaves desbalanceadas en ' + nombre);
}

class Attr {
  constructor(array, itemSize) { this.array = array; this.itemSize = itemSize; }
}
class Geometry {
  setAttribute(name, attr) { this[name] = attr; }
  setIndex(index) { this.index = { array: index }; }
  computeVertexNormals() {
    const pos = this.position.array;
    const idx = this.index.array;
    const nor = new Float32Array(pos.length);
    for (let i = 0; i < idx.length; i += 3) {
      const a = idx[i] * 3, b = idx[i + 1] * 3, c = idx[i + 2] * 3;
      const ux = pos[b] - pos[a], uy = pos[b + 1] - pos[a + 1], uz = pos[b + 2] - pos[a + 2];
      const vx = pos[c] - pos[a], vy = pos[c + 1] - pos[a + 1], vz = pos[c + 2] - pos[a + 2];
      const nx = uy * vz - uz * vy, ny = uz * vx - ux * vz, nz = ux * vy - uy * vx;
      for (const o of [a, b, c]) { nor[o] += nx; nor[o + 1] += ny; nor[o + 2] += nz; }
    }
    for (let i = 0; i < nor.length; i += 3) {
      const len = Math.hypot(nor[i], nor[i + 1], nor[i + 2]) || 1;
      nor[i] /= len; nor[i + 1] /= len; nor[i + 2] /= len;
    }
    this.normal = new Attr(nor, 3);
  }
}
const ctx = { THREE: { BufferGeometry: Geometry, Float32BufferAttribute: Attr } };
vm.createContext(ctx);
vm.runInContext(extraerFn(html, 'parseOBJ3') + '\nthis.parseOBJ3 = parseOBJ3;', ctx);
const parseOBJ3 = ctx.parseOBJ3;

function metricas(geo) {
  const pos = geo.position.array;
  const nor = geo.normal.array;
  const idx = geo.index.array;
  let volumen = 0;
  let alineadas = 0;
  let total = 0;
  for (let i = 0; i < idx.length; i += 3) {
    const a = idx[i] * 3;
    const b = idx[i + 1] * 3;
    const c = idx[i + 2] * 3;
    const ax = pos[a], ay = pos[a + 1], az = pos[a + 2];
    const bx = pos[b], by = pos[b + 1], bz = pos[b + 2];
    const cx = pos[c], cy = pos[c + 1], cz = pos[c + 2];
    volumen += (ax * (by * cz - bz * cy) - ay * (bx * cz - bz * cx) + az * (bx * cy - by * cx)) / 6;
    const ux = bx - ax, uy = by - ay, uz = bz - az;
    const vx = cx - ax, vy = cy - ay, vz = cz - az;
    const nx = uy * vz - uz * vy;
    const ny = uz * vx - ux * vz;
    const nz = ux * vy - uy * vx;
    const mx = (nor[a] + nor[b] + nor[c]) / 3;
    const my = (nor[a + 1] + nor[b + 1] + nor[c + 1]) / 3;
    const mz = (nor[a + 2] + nor[b + 2] + nor[c + 2]) / 3;
    if (nx * mx + ny * my + nz * mz > 0) alineadas++;
    total++;
  }
  return { volumen, ratio: alineadas / total };
}

const modelos = ['Artillero', 'Capitan', 'Comando', 'Espia', 'Francotirador', 'Ingeniero', 'Medico'];

for (const nombre of modelos) {
  test(nombre + ': winding hacia fuera tras reflejar X', () => {
    const txt = fs.readFileSync(path.join(root, 'assets', 'soldier', nombre + '.obj'), 'utf8');
    const nv = txt.split('\n').reduce((n, l) => n + (/^v\s/.test(l.trim()) ? 1 : 0), 0);
    const skin = { skinIndex: new Array(nv * 4).fill(0), skinWeight: new Array(nv * 4).fill(0) };
    const geo = parseOBJ3(txt, skin);
    const m = metricas(geo);
    assert.ok(m.volumen > 0, 'volumen firmado ' + m.volumen.toFixed(3));
    assert.ok(m.ratio > 0.9, 'normales alineadas ' + (m.ratio * 100).toFixed(1) + '%');
  });
}

import itertools
import json
import math
import os
import subprocess
import sys

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
PACK_PATH = os.path.join(ROOT, "assets", "soldier_clips.json")
MANIFEST_PATH = os.path.join(ROOT, "assets", "soldier_clip_manifest.json")
RIG_PATH = os.path.join(ROOT, "assets", "soldier_rig.json")


def load_json(path):
    with open(path, encoding="utf-8") as handle:
        return json.load(handle)


def git_json(commit, path):
    raw = subprocess.check_output(["git", "-C", ROOT, "show", commit + ":" + path])
    return json.loads(raw.decode("utf-8"))


def close(a, b, tolerance):
    return abs(a - b) <= tolerance


def compare_times(original, baked, tolerance):
    if len(original) != len(baked):
        return "claves: " + str(len(original)) + " vs " + str(len(baked))
    for a, b in zip(original, baked):
        if not close(a, b, tolerance):
            return "tiempo " + str(a) + " vs " + str(b)
    return None


def main():
    errors = []
    pack = load_json(PACK_PATH)
    manifest = load_json(MANIFEST_PATH)
    rig = load_json(RIG_PATH)
    bind = {bone["name"]: bone["quat"] for bone in rig["bones"]}
    entries = manifest.get("clips", [])
    keys = [entry["key"] for entry in entries]
    sources = [entry["source"] for entry in entries]
    if manifest.get("total") != 40 or len(entries) != 40:
        errors.append("el manifiesto debe cubrir 40 clips")
    if len(set(keys)) != len(keys):
        errors.append("claves de pack duplicadas")
    if len(set(sources)) != len(sources):
        errors.append("clips originales duplicados")
    if len(pack.get("bones", [])) != 20:
        errors.append("el pack debe tener 20 huesos")
    if len(pack.get("clips", {})) != 40:
        errors.append("el pack debe tener 40 clips")
    if not close(pack.get("rootScale", 0), manifest.get("rootScale", 0), 1e-6):
        errors.append("rootScale distinto entre pack y manifiesto")
    for key in keys:
        if key not in pack.get("clips", {}):
            errors.append("falta el clip " + key)
    for bone in pack.get("bones", []):
        if bone not in bind:
            errors.append("hueso sin bind: " + bone)
    commit = manifest.get("sourceCommit", "")
    source_path = manifest.get("sourcePath", "").strip("/")
    for entry in entries:
        key = entry.get("key")
        clip = pack.get("clips", {}).get(key)
        if not clip:
            continue
        if not close(clip.get("dur", 0), entry.get("duration", 0), 0.0002):
            errors.append(key + ": duracion distinta")
        for event in clip.get("events", []):
            if not isinstance(event.get("t"), (int, float)) or not isinstance(event.get("f"), str) or not isinstance(event.get("i"), int):
                errors.append(key + ": evento mal formado")
        if len(clip.get("events", [])) != entry.get("events"):
            errors.append(key + ": numero de eventos distinto")
        if key == "reference":
            for bone in pack["bones"]:
                track = clip.get("rot", {}).get(bone)
                if not track or compare_times(track.get("t", []), [0, clip["dur"]], 0.0002):
                    errors.append("reference: pista de " + bone + " no constante")
                    continue
                for quat in track.get("q", []):
                    if max(abs(a - b) for a, b in zip(quat, bind[bone])) > 1e-6:
                        errors.append("reference: bind de " + bone + " distinto")
            if clip.get("root", {}).get("p") != [[0, 0, 0], [0, 0, 0]]:
                errors.append("reference: root no neutro")
            continue
        original = git_json(commit, source_path + "/" + entry["file"])
        if not close(original["duracion_seg"], clip["dur"], 0.0002):
            errors.append(key + ": duracion original distinta")
        source_events = original.get("eventos", [])
        if len(source_events) != len(clip.get("events", [])):
            errors.append(key + ": eventos originales distintos")
        else:
            for src, dst in zip(source_events, clip["events"]):
                if not close(src["tiempo"], dst["t"], 0.0002) or src["funcion"] != dst["f"] or src["intParam"] != dst["i"]:
                    errors.append(key + ": evento " + str(dst.get("f")) + " desalineado")
                    break
        rotations = {track["hueso_corto"]: track for track in original["tracks_rotacion"]}
        for bone in pack["bones"]:
            track = clip.get("rot", {}).get(bone)
            source = rotations.get(bone)
            if not track or not source:
                errors.append(key + ": falta rotacion de " + bone)
                continue
            if compare_times(source["tiempos"], track["t"], 0.0002):
                errors.append(key + ": tiempos de " + bone + " desalineados")
            for quat in track["q"]:
                norm = math.sqrt(sum(value * value for value in quat))
                if not close(norm, 1, 0.01):
                    errors.append(key + ": cuaternion no normalizado en " + bone)
                    break
        positions = {track["hueso_corto"]: track for track in original["tracks_posicion"]}
        root_source = positions.get("Bip001")
        root = clip.get("root")
        if not root_source or not root:
            errors.append(key + ": falta root")
            continue
        if compare_times(root_source["tiempos"], root["t"], 0.0002):
            errors.append(key + ": tiempos de root desalineados")
        origin = root_source["posiciones"][0]
        scale = pack["rootScale"]
        deltas = [[value - origin[i] for i, value in enumerate(src)] for src in root_source["posiciones"]]
        candidates = []
        for axes in itertools.permutations(range(3)):
            for signs in itertools.product((-1, 1), repeat=3):
                candidates.append((axes, signs))
        matched = False
        for axes, signs in candidates:
            worst = 0
            for delta, dst in zip(deltas, root["p"]):
                expected = [signs[i] * scale * delta[axes[i]] for i in range(3)]
                worst = max(worst, max(abs(a - b) for a, b in zip(expected, dst)))
            if worst <= 0.0005:
                matched = True
                break
        if not matched and entry.get("rootTransform") == "authored":
            matched = all(all(isinstance(value, (int, float)) and math.isfinite(value) and abs(value) <= 0.5 for value in point) for point in root["p"])
            matched = matched and root["p"][0] == [0, 0, 0]
        if not matched:
            errors.append(key + ": root motion desalineado")
    if errors:
        for error in errors[:40]:
            print("ERROR", error)
        print("FALLO", len(errors), "problemas")
        return 1
    print("OK 40 clips originales, 20 huesos, eventos, tiempos, root motion y bind de referencia")
    return 0


if __name__ == "__main__":
    sys.exit(main())

=== tools\integrar_modelos_propios.py ===
"""Integra los modelos Meshy en el rig del IPA usando solo Python estandar.

Los OBJ de Meshy estan en Y arriba; el motor espera Z arriba (como los
assets del IPA). Ademas, el rig de caja generado por el pipeline no comparte
la jerarquia ni la pose de reposo del rig original, asi que los clips por
indice retorcian los brazos. Este script reutiliza el rig base del IPA:
malla rotada a Z arriba, huesos del original escalados a 1,71 m y pesos
recalculados por proximidad a los segmentos oseos.
"""

import json
import math
import os
import shutil

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
PROPIOS = os.path.join(ROOT, "assets", "personajes_propios")
SOLDADO = os.path.join(ROOT, "assets", "soldier")
CLASES = ("Artillero", "Espia", "Francotirador")


def qmul(a, b):
    x1, y1, z1, w1 = a
    x2, y2, z2, w2 = b
    return (
        w1 * x2 + x1 * w2 + y1 * z2 - z1 * y2,
        w1 * y2 - x1 * z2 + y1 * w2 + z1 * x2,
        w1 * z2 + x1 * y2 - y1 * x2 + z1 * w2,
        w1 * w2 - x1 * x2 - y1 * y2 - z1 * z2,
    )


def qmat(q):
    x, y, z, w = q
    return (
        (1 - 2 * (y * y + z * z), 2 * (x * y - z * w), 2 * (x * z + y * w)),
        (2 * (x * y + z * w), 1 - 2 * (x * x + z * z), 2 * (y * z - x * w)),
        (2 * (x * z - y * w), 2 * (y * z + x * w), 1 - 2 * (x * x + y * y)),
    )


def mv(m, v):
    return tuple(sum(m[i][j] * v[j] for j in range(3)) for i in range(3))


def world_bones(bones):
    out = []
    for b in bones:
        q = tuple(b["quat"])
        p = tuple(b["pos"])
        if b["parent"] >= 0:
            pp, pq = out[b["parent"]]
            p = tuple(pp[i] + mv(qmat(pq), p)[i] for i in range(3))
            q = qmul(pq, q)
        out.append((p, q))
    return out


def transform_obj(src, dst, altura_base):
    verts = []
    with open(src, encoding="utf8") as fi:
        alturas = [float(l.split()[2]) for l in fi if l.startswith("v ")]
    factor = altura_base / (max(alturas) - min(alturas))
    with open(src, encoding="utf8") as fi, open(dst, "w", encoding="utf8") as fo:
        for line in fi:
            p = line.split()
            if not p:
                fo.write(line)
                continue
            if p[0] == "v":
                x, y, z = float(p[1]), float(p[2]), float(p[3])
                verts.append((x * factor, -z * factor, y * factor))
                fo.write("v %.6f %.6f %.6f\n" % verts[-1])
            elif p[0] == "vn":
                x, y, z = float(p[1]), float(p[2]), float(p[3])
                fo.write("vn %.6f %.6f %.6f\n" % (x, -z, y))
            else:
                fo.write(line)
    return verts


def seg_dist(p, a, b):
    ab = tuple(b[i] - a[i] for i in range(3))
    ap = tuple(p[i] - a[i] for i in range(3))
    den = sum(v * v for v in ab)
    t = max(0.0, min(1.0, sum(ap[i] * ab[i] for i in range(3)) / den)) if den else 0.0
    return math.sqrt(sum((ap[i] - t * ab[i]) ** 2 for i in range(3)))


def make_rig(base, verts):
    with open(base, encoding="utf8") as f:
        src = json.load(f)
    ws = world_bones(src["bones"])
    pts = [tuple(p) for p, _ in ws]
    secs = []
    for i, b in enumerate(src["bones"]):
        if b["parent"] >= 0:
            secs.append((pts[b["parent"]], pts[i]))
        else:
            secs.append((pts[i], pts[i]))
    skin_bones = src["skinBones"]
    slot = {bone: i for i, bone in enumerate(skin_bones)}
    idx, wet = [], []
    for v in verts:
        ds = sorted(
            ((seg_dist(v, a, b), i) for i, (a, b) in enumerate(secs) if i in slot),
            key=lambda x: x[0],
        )[:4]
        ws4 = [1.0 / (d * d + 0.0004) for d, _ in ds]
        total = sum(ws4)
        idx.extend(slot[i] for _, i in ds)
        wet.extend(w / total for w in ws4)
    rig = {
        "bones": src["bones"],
        "skinBones": src["skinBones"],
        "skinIndex": idx,
        "skinWeight": [round(w, 6) for w in wet],
        "boneInverses": src["boneInverses"],
        "notes": "rig y bind del IPA base; clips por indice; pies a y=0",
    }
    return rig


def main():
    base = os.path.join(ROOT, "assets", "soldier_rig.json")
    zs = []
    with open(os.path.join(ROOT, "assets", "soldier.obj"), encoding="utf8") as f:
        for line in f:
            if line.startswith("v "):
                zs.append(float(line.split()[3]))
    altura_base = max(zs) - min(zs)
    for role in CLASES:
        src = os.path.join(PROPIOS, role + ".obj")
        obj = os.path.join(SOLDADO, role + ".obj")
        rig = os.path.join(SOLDADO, role + "_rig.json")
        tex = os.path.join(SOLDADO, role + "_tex.png")
        verts = transform_obj(src, obj, altura_base)
        with open(rig, "w", encoding="utf8") as f:
            json.dump(make_rig(base, verts), f, separators=(",", ":"))
        shutil.copyfile(os.path.join(PROPIOS, role + "_tex.png"), tex)
        print("%s: %d vertices, rig %d huesos, altura %.2f" % (role, len(verts), 20, altura_base))


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Suelda los OBJ Meshy por posiciÃ³n (las costuras de UV duplican vÃ©rtices),
promedia UV por grupo y guarda V/F/UV por modelo en work/weld/."""
import glob, os
import numpy as np
import trimesh

os.makedirs("weld", exist_ok=True)

def weld_by_position(path):
    # las caras traen vi/ti/ni con ti INDEPENDIENTE de vi (mÃ¡s vt que v):
    # se parsean las parejas reales y el UV de cada vÃ©rtice es el promedio
    # de los UV de las esquinas que lo referencian.
    V, UV, corners = [], [], []
    for line in open(path, encoding="utf-8", errors="ignore"):
        p = line.split()
        if not p:
            continue
        if p[0] == "v":
            V.append((float(p[1]), float(p[2]), float(p[3])))
        elif p[0] == "vt":
            UV.append((float(p[1]), float(p[2])))
        elif p[0] == "f":
            for t in p[1:4]:
                vi, ti, _ = (t.split("/") + ["", "", ""])[:3]
                corners.append((int(vi) - 1, int(ti) - 1 if ti else -1))
    V = np.array(V); UV = np.array(UV) if UV else None
    key = np.round(V, 5)
    uniq, inv = np.unique(key, axis=0, return_inverse=True)
    faces = np.array(corners, dtype=np.int64).reshape(-1, 3)
    # caras vÃ¡lidas: vi dentro de rango (descarta vÃ©rtices negativos del OBJ)
    ok = np.all(faces >= 0, axis=1) & (faces[:, 0] < len(V))
    faces = faces[ok]
    if UV is not None:
        ti = faces[:, 1]
        good = (ti >= 0) & (ti < len(UV))
        newUV = np.zeros((len(uniq), 2)); cnt = np.zeros(len(uniq))
        np.add.at(newUV, inv[faces[good, 0]], UV[ti[good]])
        np.add.at(cnt, inv[faces[good, 0]], 1)
        cnt[cnt == 0] = 1
        newUV /= cnt[:, None]
    else:
        newUV = None
    newF = inv[faces[:, 0]]
    return newF, uniq, newUV, len(V), len(uniq)

for root in sorted(glob.glob("raw/*/Meshy_*")):
    name = root.split("Meshy_AI_")[-1].split("_2026")[0].rstrip("/")
    F, V, UV, nv0, nv1 = weld_by_position(glob.glob(root + "/*.obj")[0])
    F = np.asarray(F, dtype=np.int64).reshape(-1, 3)
    np.save(f"weld/{name}_V.npy", V.astype(np.float32))
    np.save(f"weld/{name}_F.npy", F.astype(np.int32))
    if UV is not None:
        np.save(f"weld/{name}_UV.npy", UV.astype(np.float32))
    m = trimesh.Trimesh(vertices=V, faces=F, process=False)
    uniq, counts = np.unique(m.edges_sorted, axis=0, return_counts=True)
    print(f"{name}: {nv0}->{nv1} verts Â· borde {int((counts==1).sum()/2)} Â· watertight {m.is_watertight} Â· tris {len(m.faces)}")
